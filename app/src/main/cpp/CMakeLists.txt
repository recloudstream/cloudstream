cmake_minimum_required(VERSION 3.22.1)

# libass (https://github.com/libass/libass) for ASS/SSA subtitles, see LibassSubtitleSupport.kt
#
# libass, FreeType, FriBidi and HarfBuzz are built as static libraries and linked into
# libcs3ass.so. The sources are downloaded by libass-deps.cmake.
# Fontconfig is not used, fonts come from the subtitle (MKV attachments or [Fonts]) or the
# default system font.
project(cs3ass C CXX ASM)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)

# Subtitles are rendered every frame, so optimize debug builds too
add_compile_options(-O2 -ffunction-sections -fdata-sections)

include(libass-deps.cmake)

set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")

# FreeType
set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${FREETYPE_SOURCE_DIR}" freetype EXCLUDE_FROM_ALL)

# FriBidi only has a meson build, so its config headers are written here.
# The release tarball already contains the generated Unicode tables (*.tab.i).
set(FRIBIDI_GENERATED_DIR "${GENERATED_DIR}/fribidi")
file(WRITE "${FRIBIDI_GENERATED_DIR}/config.h"
    "#define HAVE_MEMMOVE 1\n"
    "#define HAVE_MEMSET 1\n"
    "#define HAVE_STRDUP 1\n"
    "#define HAVE_STDLIB_H 1\n"
    "#define HAVE_STRING_H 1\n"
    "#define HAVE_MEMORY_H 0\n"
    "#define HAVE_STRINGS_H 1\n"
    "#define HAVE_SYS_TIMES_H 1\n"
    "#define STDC_HEADERS 1\n"
    "#define HAVE_STRINGIZE 1\n"
)
string(REPLACE "." ";" FRIBIDI_VERSION_PARTS "${FRIBIDI_VERSION}")
list(GET FRIBIDI_VERSION_PARTS 0 FRIBIDI_MAJOR_VERSION)
list(GET FRIBIDI_VERSION_PARTS 1 FRIBIDI_MINOR_VERSION)
list(GET FRIBIDI_VERSION_PARTS 2 FRIBIDI_MICRO_VERSION)
set(configure_input "fribidi-config.h generated by CMake")
set(PACKAGE fribidi)
set(PACKAGE_NAME "GNU FriBidi")
set(PACKAGE_BUGREPORT "https://github.com/fribidi/fribidi/issues/new")
set(FRIBIDI_INTERFACE_VERSION 4)
set(SIZEOF_INT 4)
set(FRIBIDI_MSVC_BUILD_PLACEHOLDER "#undef FRIBIDI_BUILT_WITH_MSVC")
configure_file("${FRIBIDI_SOURCE_DIR}/lib/fribidi-config.h.in"
    "${FRIBIDI_GENERATED_DIR}/fribidi-config.h" @ONLY)

add_library(fribidi STATIC
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-arabic.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-bidi.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-bidi-types.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-char-sets.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-char-sets-cap-rtl.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-char-sets-cp1255.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-char-sets-cp1256.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-char-sets-iso8859-6.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-char-sets-iso8859-8.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-char-sets-utf8.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-deprecated.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-joining.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-joining-types.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-mirroring.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-brackets.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-run.c"
    "${FRIBIDI_SOURCE_DIR}/lib/fribidi-shape.c"
)
target_include_directories(fribidi PUBLIC "${FRIBIDI_GENERATED_DIR}" "${FRIBIDI_SOURCE_DIR}/lib")
target_compile_definitions(fribidi PRIVATE HAVE_CONFIG_H)

# HarfBuzz, single file build without any of the optional integrations
add_library(harfbuzz STATIC "${HARFBUZZ_SOURCE_DIR}/src/harfbuzz.cc")
target_include_directories(harfbuzz PUBLIC "${HARFBUZZ_SOURCE_DIR}/src")
target_compile_features(harfbuzz PRIVATE cxx_std_11)
target_compile_options(harfbuzz PRIVATE -fno-exceptions -fno-rtti -fno-threadsafe-statics)

# libass
set(LIBASS_GENERATED_DIR "${GENERATED_DIR}/libass")
set(LIBASS_CONFIG
    "#define HAVE_STRDUP 1\n"
    "#define HAVE_STRNDUP 1\n"
    "#define CONFIG_LARGE_TILES 0\n"
    "#define CONFIG_SOURCEVERSION \"cloudstream, release ${LIBASS_VERSION}\"\n"
)
set(LIBASS_SOURCES
    "${LIBASS_SOURCE_DIR}/libass/c/c_be_blur.c"
    "${LIBASS_SOURCE_DIR}/libass/c/c_blend_bitmaps.c"
    "${LIBASS_SOURCE_DIR}/libass/c/c_blur.c"
    "${LIBASS_SOURCE_DIR}/libass/c/c_rasterizer.c"
    "${LIBASS_SOURCE_DIR}/libass/ass.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_arabic_charmap.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_bitmap.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_bitmap_engine.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_blur.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_cache.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_drawing.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_filesystem.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_font.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_fontselect.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_library.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_outline.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_parse.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_rasterizer.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_render.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_render_api.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_shaper.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_string.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_strtod.c"
    "${LIBASS_SOURCE_DIR}/libass/ass_utils.c"
)
# Only the ARM64 assembly is used, the x86 one needs NASM
if(ANDROID_ABI STREQUAL "arm64-v8a")
    string(APPEND LIBASS_CONFIG "#define CONFIG_ASM 1\n#define ARCH_AARCH64 1\n")
    list(APPEND LIBASS_SOURCES
        "${LIBASS_SOURCE_DIR}/libass/aarch64/asm.S"
        "${LIBASS_SOURCE_DIR}/libass/aarch64/be_blur.S"
        "${LIBASS_SOURCE_DIR}/libass/aarch64/blend_bitmaps.S"
        "${LIBASS_SOURCE_DIR}/libass/aarch64/blur.S"
        "${LIBASS_SOURCE_DIR}/libass/aarch64/rasterizer.S"
    )
endif()
file(WRITE "${LIBASS_GENERATED_DIR}/config.h" ${LIBASS_CONFIG})

add_library(ass STATIC ${LIBASS_SOURCES})
target_include_directories(ass
    PRIVATE "${LIBASS_GENERATED_DIR}"
    PUBLIC "${LIBASS_SOURCE_DIR}/libass"
)
target_compile_definitions(ass PRIVATE _GNU_SOURCE _XPLATFORM_SOURCE)
target_compile_options(ass PRIVATE $<$<COMPILE_LANGUAGE:C>:-std=gnu99 -fno-math-errno>)
target_link_libraries(ass PRIVATE freetype fribidi harfbuzz m)

# JNI bindings
add_library(cs3ass SHARED libass_jni.c)
target_link_libraries(cs3ass PRIVATE ass freetype fribidi harfbuzz jnigraphics log m)
target_link_options(cs3ass PRIVATE
    -Wl,--gc-sections
    -Wl,--exclude-libs,ALL
    # 16 KB page size support
    -Wl,-z,max-page-size=16384
)
